\section{Algoritmo de Entrenamiento: TD3}

\begin{frame}{Twin Delayed Deep Deterministic Policy Gradient}
    \begin{itemize}
        \item Metodo actor-critico
        \item Politicas en espacios de accion continuos
        \item Mejora a su predecesor \textbf{DDPG} en:
            \begin{itemize}
                \item Sensibilidad a hiperparametros
                \item Sobreestimacion de funcion $Q$.
            \end{itemize}
    \end{itemize}

    Introduce tres estrategias:
    \begin{itemize}
        \item \textbf{Clipped Double-Q Learning}: usa dos criticos
        \item \textbf{"Delayed" Policy Updates}: la politica se actualiza con menor frecuencia que la funcion Q
        \item \textbf{Target Policy Smoothing}: agrega ruido a las acciones las \textit{target networks}
    \end{itemize}
\end{frame}

\begin{frame}{Componentes del Algoritmo}
    \begin{itemize}
        \item \textbf{Sequence Buffer} ($\mathcal{D}$): almacenamiento de transiciones alguna vez vistas durante el entrenamiento.
        \item \textbf{Target networks}: versiones rezagadas de los criticos y actor. Buscan estabilizar el entrenamiento.
        \item \textbf{Critico}: aproxima funcion de valor-accion $Q(s,a) \sim Q_{\phi}$
        \begin{equation}
            L(\phi, \mathcal{D}) = \mathbb{E}_{(s, a, r, s', d) \sim \mathcal{D}} \left[ \left( Q_{\phi}(s,a) - (r + \gamma(1-d) \max_{a'} Q_{\phi}(s', a'))\right)^2 \right]
        \end{equation}
        \item \textbf{Actor}: representa la politica del agente $\pi \sim \mu_{\theta}$
        \begin{equation}
            L(\theta, \mathcal{D}) = -\mathrm{E}_{s \sim \mathcal{D}} \left[ Q_{\phi}(s, \mu_{\theta}(s)) \right]
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{TD3+BPTT: Recoleccion de Experiencia}
    \begin{algorithmic}[1]
        \State Inicializar redes críticas $Q_{\phi_1}, Q_{\phi_2}$ y actor recurrente $\mu_\theta$.
        \State Inicializar \textit{target networks} $\phi'_1 \leftarrow \phi_1, \phi'_2 \leftarrow \phi_2, \theta' \leftarrow \theta$.
        \State Inicializar Sequence Buffer $\mathcal{D}$.
        \State Definir longitud de secuencia de entrenamiento $L$ y longitud de \textit{burn-in} (opcional).
        \For{cada paso $t = 1$ hasta $T_{ep}$}
            \State Obtener acción y nuevo estado oculto: $a_t = \mu_\theta(o_t)$.
            \State $a_t = \text{clip}(a_t + \epsilon, a_{min}, a_{max}), \quad \epsilon \sim \mathcal{N}(0, \sigma)$. \hfill \textit{\color{blue} Aplicar Exploración}
            \State Ejecutar $a_t$, observar recompensa $r_t$, nueva obs $o_{t+1}$ y bandera $d$.
            \State Almacenar transición $(o_t, a_t, r_t, o_{t+1}, d)$ en el episodio actual en $\mathcal{D}$.
            
            \State \textbf{Paso de Actualización}
        \EndFor
    \end{algorithmic}
\end{frame}

\begin{frame}{TD3+BPTT: Paso de actualización}
    \begin{algorithmic}[1]
        \State Muestrear secuencias de longitud $L$: $\tau = \{(o_k, a_k, r_k, o_{k+1}, done)\}_{k=1}^L \sim \mathcal{D}$
        \State \textbf{Paso de Crítico:}
        \State \quad $\tilde{a}_{k} = \mu_{\theta'}(o_{k+1}) + \text{clip}(\epsilon, -c, c)$ \hfill \textit{\color{blue}Target Policy Smoothing}
        \State \quad $y_{k} = r_{k} + \gamma(1-d_{k}) \min_{i=1,2} Q_{\phi'_i}(o_{k+1}, \tilde{a}_{k})$ \hfill \textit{\color{blue}Clipped Double Q-Learning}
        \State \quad Actualizar $\phi_i$ vía BPTT: $\nabla_{\phi_i} \frac{1}{L} \sum_{k=1}^L (Q_{\phi_i}(o_k, a_k) - y_k)^2$
        
        \If{Momento de actualizar Politica} \hfill \textit{\color{blue}Delayed Policy Update}
            \State \textbf{Paso de Actor:}
            \State \quad Actualizar $\theta$ vía BPTT: $\nabla_{\theta} \left( - \frac{1}{L} \sum_{k=1}^L Q_{\phi_1}(o_k, \mu_\theta(o_k)) \right)$
            \State \textbf{Actualización Target Networs:}
            \State \quad $\phi'_i \leftarrow \tau \phi_i + (1-\tau)\phi'_i, \quad \theta' \leftarrow \tau \theta + (1-\tau)\theta'$
        \EndIf
    \end{algorithmic}
\end{frame}
