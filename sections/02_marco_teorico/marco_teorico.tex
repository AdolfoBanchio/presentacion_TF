\section{Marco Teorico}

\begin{frame}{Inteligencia Artificial y ML}
  \begin{itemize}
    \item Busqueda de \textbf{agentes} capaces de tomar la decision correcta en funcion de la información disponible.
  \end{itemize}

  \begin{block}{Aprendizaje automático (\textbf{Machine Learning})}
    Un programa de computadora aprende de la experiencia E con repecto a alguna clase de tareas T y una medida de desempeño P, si su desempeño en tareas de T, medidio por P, mejora con la experiencia E.
  \end{block}
  Segun el tipo de experiencia y medida de desempeño, se divide en:
  
    \begin{itemize}
      \item Aprendizaje supervisado
      \item Aprendizaje no supervisado
      \item \textbf{Aprendizaje por refuerzo}
    \end{itemize}
\end{frame}

\begin{frame}{Aprendizaje por Refuerzo (RL)}
  \begin{columns}[T,onlytextwidth]

    % ---- Columna izquierda: texto ----
    \column{0.50\textwidth}
    Un agente interactúa con un entorno dinámico donde:
    \begin{itemize}
        \item Observa estados $s_t$
        \item Ejecuta acciones $a_t$
        \item Recibe recompensas $r_{t+1}$
    \end{itemize}

    \vspace{0.2cm}
    Con el objetivo de maximizar la recompensa final acumulada:
    \[
    \max_\pi \ \mathbb{E}[R]
    \]

    Principales desafíos:
    \begin{itemize}
        \item Problema de asignación de crédito
        \item Exploración vs explotación
        \item Retroalimentación escasa y diferida
    \end{itemize}

\begin{frame}{Procesos de Decisión de Markov (MDP)}
    La interacción \textbf{Agente–Entorno} se formaliza mediante un Proceso de Decisión de Markov (MDP).

    \begin{block}{Definición}
        Un MDP se define como la tupla $\mathcal{M} = \langle S, A, P, R, \gamma \rangle$
        donde:
        \begin{itemize}
            \item $S$: conjunto de estados posibles.
            \item $A$: conjunto de acciones disponibles.
            \item $P(s' \mid s,a)$: probabilidad de transición al estado $s'$ al ejecutar $a$ en $s$.
            \item $R(s,a,s')$: recompensa inmediata esperada asociada a la transición.
            \item $\gamma \in [0,1]$: factor de descuento.
        \end{itemize}
    \end{block}

    \begin{block}{Propiedad de Markov}
        El estado actual $s_t$ contiene toda la información necesaria para 
        predecir la evolución futura del entorno:
        \[
            \mathbb{P}(s_{t+1}, r_{t+1} \mid s_t, a_t, s_{t-1}, a_{t-1}, \dots, s_0)
            =
            \mathbb{P}(s_{t+1}, r_{t+1} \mid s_t, a_t)
        \]
    \end{block}
\end{frame}

\begin{frame}{Componentes del Aprendizaje por Refuerzo}
    El Agente toma acciones siguiendo una \textbf{Política}: mapeo de estados y observaciones del entorno a una accion que debera tomar el agente.
    
    \textbf{Objetivo del agente}: encontrar la politica que maximice el retorno final.

    \begin{block}{Retorno y Politica Optima}
        Dado una secuencia de recompensas $r_{t+1},r_{t+2}\cdots$, se define el retorno como:
        $$
        G_t = \sum_{k=0}^{T-t-1} {\gamma^k r_{t+k+1}}
        $$

        La politica optima sera $\pi^\ast = \arg\max_{\pi} J(\pi)$ donde
        $J(\pi) = \mathbb{E}[G_0]$
    \end{block}
\end{frame}

\begin{frame}{Componentes del Aprendizaje por Refuerzo}
    \begin{block}{Ecuaciones de Bellman}
        Ecuaciones recursivas que nos permiten estimar la recompensa esperada segun la situacion actual del Agente en el entorno.
        
        \begin{itemize}
            \item \textbf{Función de valor de estado}: estima la recompensa acumulada esperada al partir del estado $s$ y seguir la política $\pi$.
                \begin{itemize}
                    \item $V^\pi(s) = \mathbb{E}_\pi \left[ r_{t+1} + \gamma V^\pi(S_{t+1}) \mid S_t = s \right]$
                \end{itemize}
            \item \textbf{Función de valor de acción} estima la recompensa acumulada esperada tras ejecutar la acción $a$ en el estado $s$ y continuar siguiendo $\pi$.
                \begin{itemize}
                    \item $Q^\pi(s,a) = \mathbb{E}_\pi \left[ r_{t+1} + \gamma Q^\pi(S_{t+1}, A_{t+1}) \mid S_t = s, A_t = a \right]$.
                \end{itemize} 
        \end{itemize}
    \end{block}

    En espacios de accion continuos:
    \begin{itemize}
        \item Computar $\arg\max_a Q(s,a)$ no es factible
        \item Se aproximan dichas funciones mediante redes neuronales
    \end{itemize}
\end{frame}

\begin{frame}{Métodos Actor-Crítico}
    Se utilizan dos redes neuronales:
    \begin{itemize}
        \item \textbf{Actor}: aproxima la política
        \item \textbf{Critic}: aproxima la función de valor
    \end{itemize}

    Aproximaciones paramétricas:
    \[
    Q^\pi(s,a) \approx Q(s,a;\theta)
    \]
    \[
    \pi(a \mid s) \approx \pi(a \mid s; \phi)
    \]
    \begin{itemize}
        \item El Actor produce directamente la acción
        \item El Critic evalúa dicha acción
    \end{itemize}
\end{frame}

\begin{frame}{Descenso por el Gradiente y BPTT}
    \begin{itemize}
        \item El entrenamiento se basa en minimizar una \textbf{función de error} $L(\theta)$.
        \item El \textbf{descenso por el gradiente} ajusta los parámetros en la dirección que reduce dicho error:
    \end{itemize}

    \vspace{0.2cm}
    \[
        \theta \leftarrow \theta - \alpha \nabla_\theta L(\theta)
    \]

    \vspace{0.3cm}
    \begin{block}{Redes Recurrentes}
        \begin{itemize}
            \item La salida actual depende del estado interno previo.
            \item El error final depende de decisiones pasadas.
            \item Se utiliza \textbf{Backpropagation Through Time (BPTT)}:
            \begin{itemize}
                \item Se despliega la red en el tiempo.
                \item El gradiente se propaga hacia atrás a lo largo de la secuencia.
            \end{itemize}
        \end{itemize}
    \end{block}
\end{frame}

    \begin{itemize}

            \begin{itemize}

            \end{itemize}
        \item Tipos de sinapsis:
            \begin{itemize}
                \item Exitatorias
                \item Inhibitorias
                \item Uniones electricas (Gap Junctions)
            \end{itemize}
    \end{itemize}
